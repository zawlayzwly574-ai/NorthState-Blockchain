  return (
    <div className="w-full font-sans text-white">
      
      {/* ==================== SPOT TRADE CONTAINER ==================== */}
      <div className="bg-[#12141D] p-5 rounded-2xl border border-slate-800 space-y-4 shadow-xl relative">
        
        {/* HEADER: SPOT TRADE -> Active Symbol Header & Futures Popup Button */}

          <div>
            <h3 className="text-lg font-bold text-amber-400 tracking-wider uppercase">{symbol}</h3>

          <div className="flex items-center gap-2">
            <input
              type="number"
              value={tradeAmount}
              onChange={(e) => setTradeAmount(e.target.value)}

            />
            <div className="grid grid-cols-2 gap-1.5">

                  className="px-2.5 py-1 text-xs font-bold rounded-lg bg-[#1A1D2A] border border-slate-700/60 hover:border-amber-500 text-slate-300 hover:text-white transition-all"
                >
                  {amt}
                </button>
              ))}
            </div>
          </div>
          <p className="text-[11px] text-slate-400 mt-1.5">

            <div className="flex items-center gap-2 text-sm font-bold">

            <span className="font-bold text-amber-400">30%</span>
          </div>
          <div>
            <span className="text-slate-400 block text-[10px] uppercase">IF WIN</span>
            <span className="font-bold text-emerald-400">+$1500</span>
          </div>
          <div>
            <span className="text-slate-400 block text-[10px] uppercase">IF LOSS</span>
            <span className="font-bold text-rose-400">-$5000</span>
          </div>
        </div>

          </button>
        </div>
      </div>


      {/* ==================== FUTURES TRADE POPUP / MODAL ==================== */}
      {isFuturesOpen && (

              </button>
              <button
                onClick={() => setFuturesDirection('Short')}

                  futuresDirection === 'Short' ? 'bg-rose-600 text-white' : 'text-slate-400 hover:text-white'

              >
                📉 Short
              </button>

            {/* Leverage Pills Sync */}
            <div>
              <div className="flex justify-between items-center text-xs text-slate-400 mb-1.5">

              <div className="flex gap-1.5 overflow-x-auto pb-1">

                  >
                    {lev}x
                  </button>

                  <span>Available: ${availableBalance.toLocaleString()}</span>
                  <button
                    onClick={() => setFuturesMargin(availableBalance.toString())}
                    className="text-[10px] font-bold text-purple-400 bg-purple-500/20 px-1.5 py-0.5 rounded hover:bg-purple-500/30"
                  >
                    MAX
                  </button>
                </div>
              </div>
              <div className="relative">
                <input
                  type="number"
                  value={futuresMargin}
                  onChange={(e) => setFuturesMargin(e.target.value)}
                  className="w-full bg-[#1A1D2A] border border-slate-700/80 rounded-xl px-4 py-2.5 text-base font-bold text-white focus:outline-none focus:border-purple-500/50"
                />
                <span className="absolute right-4 top-3 text-xs font-bold text-slate-500">{futuresSettlement}</span>
              </div>
            </div>

            {/* Risk Warning Box */}
            <div className="bg-amber-500/10 border border-amber-500/30 p-3 rounded-xl flex items-start gap-2.5 text-amber-300 text-[11px] leading-relaxed">
              <ShieldAlert className="w-4 h-4 text-amber-400 shrink-0 mt-0.5" />
              <span>
                Futures trading carries significant risk. Leverage amplifies both gains and losses. You may lose more than your initial margin.
              </span>
            </div>

            {/* Dynamic Action Button */}
            <button
              className={`w-full py-3.5 rounded-xl font-bold text-sm shadow-lg transition-all text-white ${
                futuresDirection === 'Long'
                  ? 'bg-emerald-600 hover:bg-emerald-500 shadow-emerald-600/20'
                  : 'bg-rose-600 hover:bg-rose-500 shadow-rose-600/20'
              }`}
            >
              Open {futuresLeverage}x {futuresDirection} — {symbol.split('/')[0]}/{futuresSettlement}
            </button>

          </div>
        </div>
      )}

    </div>
  );
};
